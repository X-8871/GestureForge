+def hash_variants(path: Path) -> set[str]:
+    """同一文件在两种换行约定下的可比哈希集合（O1 口径）。

+    包含三个方向，任一声明值命中即视为一致：

+    * CRLF→LF（git 归一化；`git archive`/LF 工作副本的口径）；
+    * LF→CRLF（清单按工作区 CRLF 声明、而导出为 LF 时的反方向）。

+        variants.add(hashlib.sha256(lf.replace(b"\n", b"\r\n")).hexdigest())
+    return variants
